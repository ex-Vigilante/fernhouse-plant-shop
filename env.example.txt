# Fernhouse environment variables.
# Copy these names into Vercel: Project -> Settings -> Environment Variables.
# For local work with `vercel dev` or a build you can also put them in a file called .env (never commit it).
# Leave the two VITE_ values empty and the site runs as the public demo (no payments, no database).

# Public: your Supabase project URL. Baked into the website at build time.
VITE_SUPABASE_URL=https://your-project-ref.supabase.co

# Public: the Supabase "anon" key. Safe in the browser because every table is locked down by Row Level Security.
VITE_SUPABASE_ANON_KEY=your-anon-key

# Server only: the same Supabase project URL, used by the /api functions.
SUPABASE_URL=https://your-project-ref.supabase.co

# SECRET, server only: Supabase "service_role" key. Bypasses Row Level Security. Never put it in VITE_ variables or in code.
SUPABASE_SERVICE_ROLE_KEY=your-service-role-key

# SECRET, server only: Stripe secret key (sk_test_... while testing, sk_live_... when you go live).
STRIPE_SECRET_KEY=sk_test_xxx

# SECRET, server only: signing secret of the Stripe webhook endpoint (whsec_...), used to prove events really come from Stripe.
STRIPE_WEBHOOK_SECRET=whsec_xxx

# REQUIRED, server only: your shop's public address with no trailing slash, e.g. https://fernhouse.vercel.app. Used for Stripe's return links and as the only host allowed to start a checkout.
SITE_URL=https://your-domain.example
